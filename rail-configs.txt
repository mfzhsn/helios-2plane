configure_rail() {
    local dev="$1" addr="$2" gw="$3" table="$4" priority="$5"

    ip link set "$dev" up
    ip -6 addr replace "$addr/96" dev "$dev"
    ip -6 route replace default via "$gw" dev "$dev" table "$table"
